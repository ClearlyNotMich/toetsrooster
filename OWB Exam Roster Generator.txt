function main(workbook: ExcelScript.Workbook): void {
    const bladen: Map<string, Matrix> = new Map<string, Matrix>();
    for (const ws of workbook.getWorksheets()) {
        bladen.set(ws.getName(), leesBlad(ws));
    }
    const r: Resultaat = maakRooster(bladen, MODUS);
    // Gewone for-lussen rond Excel-aanroepen (Office Scripts zet die om naar async-code).
    const namen: string[] = [];
    r.bladen.forEach((_: Matrix, naam: string) => { namen.push(naam); });
    for (const naam of namen) {
        schrijfBlad(workbook, naam, r.bladen.get(naam) as Matrix);
    }
    const toon: ExcelScript.Worksheet | undefined = workbook.getWorksheet(r.fouten.length > 0 ? r.meldingen : r.rooster);
    if (toon) {
        toon.activate();
    }
    const meldingen: Matrix = r.bladen.get(r.meldingen) as Matrix;
    for (const rij of meldingen.slice(1)) {
        console.log(`${rij[0]}: ${rij[1]}`);
    }
}

// ============================================================ Excel lezen/schrijven

function leesBlad(ws: ExcelScript.Worksheet): Matrix {
    const gebruikt: ExcelScript.Range | undefined = ws.getUsedRange(true);
    if (!gebruikt) {
        return [];
    }
    // Vanaf A1 lezen, zodat kolomnummers kloppen ook als kolom A leeg is.
    const rijen: number = gebruikt.getRowIndex() + gebruikt.getRowCount();
    const kolommen: number = gebruikt.getColumnIndex() + gebruikt.getColumnCount();
    return ws.getRangeByIndexes(0, 0, rijen, kolommen).getValues() as Matrix;
}

// Het studentenblad herken je aan de kolom EersteDatum; daarvan worden alleen de Eerste*-kolommen geschreven.
function isStudentenblad(rijen: Matrix): boolean {
    return rijen.length > 0 && rijen[0].some((k: Cel) => norm(k) === "eerstedatum");
}

function zelfdeKop(rij: Cel[], kop: string[]): boolean {
    return rij.length === kop.length && rij.every((k: Cel, i: number) => k === kop[i]);
}

function schrijfBlad(workbook: ExcelScript.Workbook, naam: string, rijen: Matrix): void {
    let ws: ExcelScript.Worksheet | undefined = workbook.getWorksheet(naam);
    if (isStudentenblad(rijen) && ws) {
        // Andere cellen (en formules) blijven ongemoeid.
        const eerste: string[] = EERSTE_KOP.map((k: string) => norm(k));
        for (let c: number = 0; c < rijen[0].length; c++) {
            if (eerste.indexOf(norm(rijen[0][c])) >= 0) {
                ws.getRangeByIndexes(0, c, rijen.length, 1).setValues(rijen.map((rij: Cel[]) => [rij[c]]));
                zetOpmaak(ws, rijen[0][c], c, rijen.length);
            }
        }
        return;
    }
    if (!ws) {
        ws = workbook.addWorksheet(naam);
    }
    const oud: ExcelScript.Range | undefined = ws.getUsedRange();
    if (oud) {
        oud.clear(ExcelScript.ClearApplyTo.contents);
    }
    const breedte: number = Math.max(...rijen.map((rij: Cel[]) => rij.length));
    const vol: Matrix = rijen.map((rij: Cel[]) => rij.concat(new Array<Cel>(breedte - rij.length).fill("")));
    const bereik: ExcelScript.Range = ws.getRangeByIndexes(0, 0, vol.length, breedte);
    bereik.setValues(vol);
    ws.getRangeByIndexes(0, 0, 1, breedte).getFormat().getFont().setBold(true);
    if (zelfdeKop(vol[0], INSTELLINGEN_KOP)) {
        const tijden: string[] = INSTELLINGEN.filter((i: Instelling) => i.soort === "tijd").map((i: Instelling) => norm(i.label));
        for (let r: number = 1; r < vol.length; r++) {
            if (tijden.indexOf(norm(vol[r][0])) >= 0) {
                ws.getRangeByIndexes(r, 1, 1, 1).setNumberFormat("hh:mm");
            }
        }
    } else if (!zelfdeKop(vol[0], MELDINGEN_KOP)) {
        for (let c: number = 0; c < breedte; c++) {
            zetOpmaak(ws, vol[0][c], c, vol.length);
        }
    }
    bereik.getFormat().autofitColumns();
}

function zetOpmaak(ws: ExcelScript.Worksheet, titel: Cel, c: number, rijen: number): void {
    const fmt: string | undefined = OPMAAK.get(norm(titel));
    if (fmt && rijen > 1) {
        ws.getRangeByIndexes(1, c, rijen - 1, 1).setNumberFormat(fmt);
    }
}

// ============================================================ hulpjes

function tekst(v: Cel | undefined | null): string {
    if (v === undefined || v === null || v === "") {
        return "";
    }
    if (typeof v === "boolean") {
        return v ? "true" : "false";
    }
    return String(v).trim();
}

function norm(v: Cel | undefined): string {
    return tekst(v).toLowerCase().replace(/[^a-z0-9]/g, "");
}

function isGetal(v: Cel): boolean {
    return typeof v === "number";
}

function waar(v: Cel): boolean {
    if (v === true || (isGetal(v) && v === 1)) {
        return true;
    }
    const t: string = tekst(v).toLowerCase();
    return t.indexOf("ja") === 0 || t === "x" || t === "waar" || t === "true";
}

const NUL: number = Date.UTC(1899, 11, 30);

function serie(j: number, m: number, d: number): number | null {
    if (j < 1) {
        return null;
    }
    const x: Date = new Date(0);
    x.setUTCFullYear(j, m - 1, d);
    if (x.getUTCFullYear() !== j || x.getUTCMonth() !== m - 1 || x.getUTCDate() !== d) {
        return null;
    }
    return Math.round((x.getTime() - NUL) / 86400000);
}

function naarDatum(v: Cel): number | null {
    if (typeof v === "number") {
        return v >= 1 ? Math.floor(v) : null;
    }
    const t: string = tekst(v);
    let m: RegExpMatchArray | null = t.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
    if (m) {
        return serie(Number(m[1]), Number(m[2]), Number(m[3]));
    }
    m = t.match(/^(\d{1,2})[-/.](\d{1,2})[-/.](\d{4})$/);
    if (m) {
        return serie(Number(m[3]), Number(m[2]), Number(m[1]));
    }
    return null;
}

function naarMinuten(v: Cel): number | null {
    if (typeof v === "number") {
        return Math.floor((v - Math.floor(v)) * 1440 + 0.5);
    }
    const m: RegExpMatchArray | null = tekst(v).match(/^(\d{1,2})[:.](\d{2})$/);
    if (m && Number(m[1]) < 24 && Number(m[2]) < 60) {
        return Number(m[1]) * 60 + Number(m[2]);
    }
    return null;
}

function twee(n: number): string {
    return (n < 10 ? "0" : "") + String(n);
}

function datumTekst(d: number): string {
    const x: Date = new Date(NUL + d * 86400000);
    return `${twee(x.getUTCDate())}-${twee(x.getUTCMonth() + 1)}-${x.getUTCFullYear()}`;
}

function tijdTekst(minuten: number): string {
    return `${twee(Math.floor(minuten / 60))}:${twee(minuten % 60)}`;
}

// Kolomnummer op kopnaam: eerst exact, anders de eerste kop die ermee begint ('Klas p2' -> klas).
function kolom(kop: Cel[], namen: string[]): number {
    const genormd: string[] = kop.map((k: Cel) => norm(k));
    for (const n of namen) {
        if (genormd.indexOf(n) >= 0) {
            return genormd.indexOf(n);
        }
    }
    for (const n of namen) {
        for (let i: number = 0; i < genormd.length; i++) {
            if (genormd[i].indexOf(n) === 0) {
                return i;
            }
        }
    }
    return -1;
}

function cel(rij: Cel[], i: number): Cel {
    return i >= 0 && i < rij.length ? rij[i] : "";
}

function namenLijst(v: Cel): string[] {
    return tekst(v).split(/[,;\/]/).map((x: string) => x.trim()).filter((x: string) => x.length > 0);
}

function beperkt(naam: string, verboden: string[], deels: boolean): boolean {
    const n: string = naam.trim().toLowerCase();
    return verboden.some((v: string) => {
        const w: string = v.toLowerCase();
        return n === w || (deels && (n.indexOf(w) >= 0 || w.indexOf(n) >= 0));
    });
}

function leeg(rij: Cel[]): boolean {
    return rij.every((x: Cel) => tekst(x) === "");
}

// Woord aan het begin van een woord in t ('laat' in 'laatste', niet in 'plaatsen').
function heeftWoord(t: string, woord: string): boolean {
    return new RegExp("(^|[^a-z])" + woord).test(t.toLowerCase());
}

function voorkeurIn(v: Cel): string {
    const gevonden: string[] = VOORKEUREN.filter((w: string) => heeftWoord(tekst(v), w.toLowerCase()));
    return gevonden.length > 0 ? gevonden[0] : "";
}

// '11:15-11:40, 13:00-13:55' -> [[675, 700], [780, 835]]; null als het niet klopt.
function leesPauzes(v: Cel): number[][] | null {
    const lijst: number[][] = [];
    for (const deel of tekst(v).split(/[,;]/)) {
        if (!deel.trim()) {
            continue;
        }
        const m: RegExpMatchArray | null = deel.trim().match(/^(\d{1,2}[:.]\d{2})\s*[-–]\s*(\d{1,2}[:.]\d{2})$/);
        const a: number | null = m ? naarMinuten(m[1]) : null;
        const b: number | null = m ? naarMinuten(m[2]) : null;
        if (a === null || b === null || a >= b) {
            return null;
        }
        lijst.push([a, b]);
    }
    lijst.sort((x: number[], y: number[]) => vergelijk(x, y));
    for (let i: number = 1; i < lijst.length; i++) {
        if (lijst[i][0] < lijst[i - 1][1]) {
            return null;
        }
    }
    return lijst;
}

function pauzesTekst(pz: number[][]): string {
    return pz.map((p: number[]) => `${tijdTekst(p[0])}-${tijdTekst(p[1])}`).join(", ");
}

// Vergelijkt sorteersleutels zoals Python tuples doet.
function vergelijk(a: (string | number)[], b: (string | number)[]): number {
    for (let i: number = 0; i < a.length; i++) {
        if (a[i] < b[i]) {
            return -1;
        }
        if (a[i] > b[i]) {
            return 1;
        }
    }
    return 0;
}

// ============================================================ inlezen

function toelichting(i: Instelling): string {
    return i.uitleg + (i.opties.length > 0 ? ` Keuzes: ${i.opties.join(" / ")}.` : "");
}

function instellingenMatrix(): Matrix {
    return [INSTELLINGEN_KOP as Cel[]].concat(INSTELLINGEN.map((i: Instelling) => [i.label, i.standaard, toelichting(i)] as Cel[]));
}

// Blad Instellingen met ontbrekende rijen toegevoegd, of null als er niets ontbreekt.
function instellingenAangevuld(matrix: Matrix | undefined): Matrix | null {
    if (!matrix || matrix.length === 0) {
        return instellingenMatrix();
    }
    const aanwezig: string[] = matrix.slice(1).map((rij: Cel[]) => norm(cel(rij, 0)));
    const nieuw: Matrix = INSTELLINGEN.filter((i: Instelling) => aanwezig.indexOf(norm(i.label)) < 0)
        .map((i: Instelling) => [i.label, i.standaard, toelichting(i)] as Cel[]);
    return nieuw.length > 0 ? matrix.map((r: Cel[]) => r.slice()).concat(nieuw) : null;
}

// w per sleutel: getal en tijd (minuten) als number, keuze/tekst als string; plus duur, ttv. Pauzes apart.
function leesInstellingen(matrix: Matrix | undefined, fouten: string[]): Inst {
    const w: Map<string, Cel> = new Map<string, Cel>();
    INSTELLINGEN.forEach((i: Instelling) => w.set(norm(i.label), i.standaard));
    (matrix || []).slice(1).forEach((rij: Cel[]) => {
        if (w.has(norm(cel(rij, 0)))) {
            w.set(norm(cel(rij, 0)), cel(rij, 1));
        }
    });
    const s: Map<string, Cel> = new Map<string, Cel>();
    let pauzes: number[][] = [];
    for (const i of INSTELLINGEN) {
        const v: Cel = w.get(norm(i.label)) as Cel;
        if (i.soort === "getal") {
            const g: number = /^\d+$/.test(tekst(v)) ? parseInt(tekst(v), 10) : -1;
            s.set(i.sleutel, g);
            if (g < i.min || g > i.max) {
                fouten.push(`Instellingen: '${i.label}' moet een geheel getal van ${i.min} t/m ${i.max} zijn.`);
                s.set(i.sleutel, i.standaard);
            }
        } else if (i.soort === "tijd") {
            let t: number | null = naarMinuten(v);
            if (t === null) {
                fouten.push(`Instellingen: '${i.label}' moet een tijd zijn (bijv. 09:40).`);
                t = naarMinuten(i.standaard) as number;
            }
            s.set(i.sleutel, t);
        } else if (i.soort === "pauzes") {
            const pz: number[][] | null = leesPauzes(v);
            if (pz === null) {
                fouten.push(`Instellingen: '${i.label}' moet zoiets zijn als '11:15-11:40, 13:00-13:55' ` +
                    "(van vóór tot, niet overlappend) of leeg.");
            }
            pauzes = pz || [];
            s.set(i.sleutel, "");
        } else if (i.soort === "keuze") {
            const gevonden: string[] = i.opties.filter((o: string) => norm(o) === norm(v));
            s.set(i.sleutel, gevonden.length > 0 ? gevonden[0] : i.standaard);
            if (gevonden.length === 0) {
                fouten.push(`Instellingen: '${i.label}' moet een van deze waarden zijn: ${i.opties.join(" / ")}.`);
            }
        } else {
            s.set(i.sleutel, tekst(v) || i.standaard);
            if (!tekst(v)) {
                fouten.push(`Instellingen: '${i.label}' mag niet leeg zijn.`);
            }
        }
    }
    s.set("duur", s.get(`duur${s.get("jaar")}`) as Cel);
    s.set("ttv", s.get(`ttv${s.get("jaar")}`) as Cel);
    if ((s.get("start") as number) >= (s.get("eind") as number)) {
        fouten.push("Instellingen: 'Eerste toets om' moet vóór 'Laatste toets klaar om' liggen.");
    }
    return { w: s, pauzes: pauzes };
}

function leesStudenten(matrix: Matrix | undefined, blad: string, fouten: string[], meldingen: Cel[][]): Student[] {
    if (!matrix || matrix.length === 0) {
        fouten.push(`Blad '${blad}' ontbreekt of is leeg.`);
        return [];
    }
    const kop: Cel[] = matrix[0];
    const kNr: number = kolom(kop, ["studentnummer", "stnr"]);
    const kVoornaam: number = kolom(kop, ["voornaam", "roepnaam"]);
    const kTv: number = kolom(kop, ["tussenvoegsel", "vv"]);
    const kAchternaam: number = kolom(kop, ["achternaam"]);
    const kKlas: number = kolom(kop, ["klas"]);
    const kSb: number = kolom(kop, ["nietinroosterenbijsbdocent"]);
    const kVd: number = kolom(kop, ["nietinroosterenbijdevaardighedendocent", "nietinroosterenbijvaardighedendocent"]);
    const kTtv: number = kolom(kop, ["ttv"]);
    const kHer: number = kolom(kop, ["isherkanser", "herkanser"]);
    const kEb: number = kolom(kop, ["eerstebeoordelaar"]);
    const kEc: number = kolom(kop, ["eerstecasus"]);
    const kVk: number = kolom(kop, ["voorkeur"]);
    const kOpm: number = kolom(kop, ["opmerkingen", "opmerking"]);
    if (kVoornaam < 0) {
        fouten.push(`Blad '${blad}': kolom 'Voornaam' ontbreekt.`);
    }
    if (kAchternaam < 0) {
        fouten.push(`Blad '${blad}': kolom 'Achternaam' ontbreekt.`);
    }
    if (fouten.length > 0) {
        return [];
    }
    const lijst: Student[] = [];
    const oud: string[] = [];
    for (let i: number = 1; i < matrix.length; i++) {
        const rij: Cel[] = matrix[i];
        if (tekst(cel(rij, kVoornaam)) === "" && tekst(cel(rij, kAchternaam)) === "") {
            continue;
        }
        const her: Cel = cel(rij, kHer);
        if (typeof her === "number" && her > 1) {
            oud.push(String(i + 1));
        }
        const st: Student = {
            i: i, nr: tekst(cel(rij, kNr)), voornaam: tekst(cel(rij, kVoornaam)), tv: tekst(cel(rij, kTv)),
            achternaam: tekst(cel(rij, kAchternaam)), klas: tekst(cel(rij, kKlas)),
            verboden: namenLijst(cel(rij, kSb)).concat(namenLijst(cel(rij, kVd))),
            ttv: waar(cel(rij, kTtv)), her: waar(her), eb: tekst(cel(rij, kEb)), ec: tekst(cel(rij, kEc)),
            voorkeur: voorkeurIn(cel(rij, kVk)), opm: tekst(cel(rij, kOpm)),
        };
        const naam: string = `${st.voornaam} ${st.achternaam}`;
        if (tekst(cel(rij, kVk)) && !st.voorkeur) {
            meldingen.push(["Waarschuwing", `${naam}: voorkeur '${tekst(cel(rij, kVk))}' onbekend ` +
                `(gebruik ${VOORKEUREN.join(" / ")}); genegeerd.`]);
        }
        if (st.opm) {
            const gelezen: string[] = (heeftWoord(st.opm, "ttv") ? ["TTV"] : [])
                .concat(voorkeurIn(st.opm) ? [`voorkeur ${voorkeurIn(st.opm).toLowerCase()}`] : []);
            st.ttv = st.ttv || heeftWoord(st.opm, "ttv");
            st.voorkeur = st.voorkeur || voorkeurIn(st.opm);
            meldingen.push(["Info", `Opmerking bij ${naam}: '${st.opm}' — ` + (gelezen.length > 0
                ? `gelezen als ${gelezen.join(" en ")}; controleer dit.` : "niet automatisch verwerkt; controleer dit zelf.")]);
        }
        lijst.push(st);
    }
    if (oud.length > 0) {
        meldingen.push(["Waarschuwing", `${blad} rij ${oud.join(", ")}: 'IsHerkanser' bevat een getal/datum ` +
            "(restant van de oude scriptfout) en wordt gelezen als 'Nee'. Maak deze cellen leeg."]);
    }
    return lijst;
}

function leesSlots(matrix: Matrix | undefined, blad: string, fouten: string[]): Slot[] {
    if (!matrix || matrix.length === 0) {
        fouten.push(`Blad '${blad}' ontbreekt of is leeg.`);
        return [];
    }
    const kop: Cel[] = matrix[0];
    const kd: number = kolom(kop, ["datum"]);
    const kt: number = kolom(kop, ["tijd"]);
    const kl: number = kolom(kop, ["lokaal"]);
    const kc: number = kolom(kop, ["casus"]);
    const kv: number = kolom(kop, ["vaardigheid"]);
    const kn: number = kolom(kop, ["casusnapauze"]);
    if (Math.min(kd, kt, kl) < 0) {
        fouten.push(`Blad '${blad}': kolommen 'Datum', 'Tijd' en 'Lokaal' zijn verplicht.`);
        return [];
    }
    const slots: Slot[] = [];
    for (let i: number = 1; i < matrix.length; i++) {
        const rij: Cel[] = matrix[i];
        if (leeg(rij)) {
            continue;
        }
        const d: number | null = naarDatum(cel(rij, kd));
        const t: number | null = naarMinuten(cel(rij, kt));
        if (d === null || t === null || tekst(cel(rij, kl)) === "") {
            fouten.push(`Blad '${blad}' rij ${i + 1}: ongeldige of lege datum, tijd of lokaal.`);
            continue;
        }
        slots.push({
            n: slots.length, d: d, t: t, lokaal: tekst(cel(rij, kl)), casus: tekst(cel(rij, kc)),
            casusNa: tekst(cel(rij, kn)), vaardigheid: tekst(cel(rij, kv)),
        });
    }
    return slots;
}

function leesBeschikbaar(matrix: Matrix | undefined, blad: string, namen: string[], fouten: string[]): Map<string, Persoon[]> {
    const beschikbaar: Map<string, Persoon[]> = new Map<string, Persoon[]>();
    if (!matrix || matrix.length === 0) {
        fouten.push(`Blad '${blad}' ontbreekt of is leeg.`);
        return beschikbaar;
    }
    const kop: Cel[] = matrix[0];
    const kd: number = kolom(kop, ["datum"]);
    const kt: number = kolom(kop, ["tijd"]);
    const kn: number = kolom(kop, namen);
    const kc: number = kolom(kop, ["casus"]);
    const kl: number = kolom(kop, ["lokaal"]);
    if (Math.min(kd, kt, kn) < 0) {
        fouten.push(`Blad '${blad}': kolommen 'Datum', 'Tijd' en de naamkolom zijn verplicht.`);
        return beschikbaar;
    }
    for (let i: number = 1; i < matrix.length; i++) {
        const rij: Cel[] = matrix[i];
        const naam: string = tekst(cel(rij, kn));
        if (naam === "") {
            continue;
        }
        const d: number | null = naarDatum(cel(rij, kd));
        const t: number | null = naarMinuten(cel(rij, kt));
        if (d === null || t === null) {
            fouten.push(`Blad '${blad}' rij ${i + 1}: ongeldige datum of tijd.`);
            continue;
        }
        const sleutel: string = `${d}_${t}`;
        if (!beschikbaar.has(sleutel)) {
            beschikbaar.set(sleutel, []);
        }
        (beschikbaar.get(sleutel) as Persoon[]).push({ naam: naam, casus: tekst(cel(rij, kc)), lokaal: tekst(cel(rij, kl)) });
    }
    return beschikbaar;
}

// Begintijden van de dag: elke toetsduur + wisseltijd een student; na een pauze weer na de wisseltijd.
function tijdgrid(start: number, eind: number, duur: number, wissel: number, pz: number[][]): number[] {
    const tijden: number[] = [];
    let t: number = start;
    while (t + duur <= eind) {
        const botsing: number[][] = pz.filter((p: number[]) => t < p[1] && t + duur > p[0]);
        if (botsing.length > 0) {
            t = botsing[0][1] + wissel;
            continue;
        }
        tijden.push(t);
        t += duur + wissel;
    }
    return tijden;
}

function heeftData(matrix: Matrix | undefined): boolean {
    return !!matrix && matrix.length > 0 && matrix.slice(1).some((r: Cel[]) => !leeg(r));
}

// Blad Toetsdag (één rij per lokaal per dag) -> slots en beschikbare beoordelaars per slot.
function leesToetsdag(matrix: Matrix, blad: string, tijden: number[], fouten: string[], slots: Slot[], beo: Map<string, Persoon[]>): void {
    const kop: Cel[] = matrix[0];
    const kd: number = kolom(kop, ["datum"]);
    const kl: number = kolom(kop, ["lokaal"]);
    const kb: number = kolom(kop, ["beoordelaar"]);
    const kc: number = kolom(kop, ["casus"]);
    const kn: number = kolom(kop, ["casusnapauze"]);
    const kv: number = kolom(kop, ["vaardigheid"]);
    const kb2: number = kolom(kop, ["beoordelaar2"]);
    const kw: number = kolom(kop, ["wisselom"]);
    if (Math.min(kd, kl, kb) < 0) {
        fouten.push(`Blad '${blad}': kolommen 'Datum', 'Lokaal' en 'Beoordelaar' zijn verplicht.`);
        return;
    }
    const gezien: string[] = [];
    for (let i: number = 1; i < matrix.length; i++) {
        const rij: Cel[] = matrix[i];
        if (leeg(rij)) {
            continue;
        }
        const d: number | null = naarDatum(cel(rij, kd));
        const lokaal: string = tekst(cel(rij, kl));
        const naam: string = tekst(cel(rij, kb));
        const naam2: string = tekst(cel(rij, kb2));
        const w: number | null = tekst(cel(rij, kw)) ? naarMinuten(cel(rij, kw)) : null;
        if (d === null || !lokaal || !naam) {
            fouten.push(`Blad '${blad}' rij ${i + 1}: vul een geldige datum, een lokaal en een beoordelaar in.`);
            continue;
        }
        if ((naam2 !== "") !== (w !== null)) {
            fouten.push(`Blad '${blad}' rij ${i + 1}: vul 'Beoordelaar 2' en 'Wissel om' (een tijd) allebei in, ` +
                "of laat beide leeg.");
            continue;
        }
        if (gezien.indexOf(`${d}_${norm(lokaal)}`) >= 0) {
            fouten.push(`Blad '${blad}' rij ${i + 1}: lokaal ${lokaal} staat al eerder op ${datumTekst(d)}.`);
            continue;
        }
        gezien.push(`${d}_${norm(lokaal)}`);
        for (const t of tijden) {
            slots.push({
                n: slots.length, d: d, t: t, lokaal: lokaal, casus: tekst(cel(rij, kc)),
                casusNa: tekst(cel(rij, kn)), vaardigheid: tekst(cel(rij, kv)),
            });
            beo.set(`${d}_${t}_${norm(lokaal)}`, [{ naam: naam2 && w !== null && t >= w ? naam2 : naam, casus: "", lokaal: "" }]);
        }
    }
}

// Blad Simu zonder kolom Tijd: één rij per simucliënt per dag (Datum leeg = elke dag), met Vanaf/Tot.
function leesSimuDag(matrix: Matrix, blad: string, fouten: string[]): SimuDag[] {
    const kop: Cel[] = matrix[0];
    const kn: number = kolom(kop, ["simuclint", "simu", "simulant", "naam"]);
    const kd: number = kolom(kop, ["datum"]);
    const kv: number = kolom(kop, ["vanaf", "van"]);
    const kt: number = kolom(kop, ["tot"]);
    const lijst: SimuDag[] = [];
    if (kn < 0) {
        fouten.push(`Blad '${blad}': kolom 'simucliënt' ontbreekt.`);
        return lijst;
    }
    for (let i: number = 1; i < matrix.length; i++) {
        const rij: Cel[] = matrix[i];
        const naam: string = tekst(cel(rij, kn));
        if (!naam) {
            continue;
        }
        const d: number | null = tekst(cel(rij, kd)) ? naarDatum(cel(rij, kd)) : null;
        const van: number | null = tekst(cel(rij, kv)) ? naarMinuten(cel(rij, kv)) : null;
        const tot: number | null = tekst(cel(rij, kt)) ? naarMinuten(cel(rij, kt)) : null;
        if ((tekst(cel(rij, kd)) && d === null) || (tekst(cel(rij, kv)) && van === null) || (tekst(cel(rij, kt)) && tot === null)) {
            fouten.push(`Blad '${blad}' rij ${i + 1}: ongeldige datum, Vanaf of Tot.`);
            continue;
        }
        lijst.push({ naam: naam, d: d, van: van, tot: tot });
    }
    return lijst;
}

// Simu per slot: in blok b (na b pauzes) krijgt lokaal k van die dag simu (k + b) mod n.
function simuRotatie(slots: Slot[], simus: SimuDag[], pz: number[][], duur: number, rotatie: boolean, meldingen: Cel[][]): Map<string, Persoon[]> {
    const blok = (t: number): number => pz.filter((p: number[]) => p[1] <= t).length;
    const kamers: Map<number, string[]> = new Map<number, string[]>();
    const bereik: Map<string, Bereik> = new Map<string, Bereik>();
    for (const slot of slots) {
        if (!kamers.has(slot.d)) {
            kamers.set(slot.d, []);
        }
        const lijst: string[] = kamers.get(slot.d) as string[];
        if (lijst.indexOf(norm(slot.lokaal)) < 0) {
            lijst.push(norm(slot.lokaal));
        }
        const sleutel: string = `${slot.d}_${blok(slot.t)}`;
        if (!bereik.has(sleutel)) {
            bereik.set(sleutel, { d: slot.d, b: blok(slot.t), a: slot.t, e: slot.t + duur, ks: [] });
        }
        const x: Bereik = bereik.get(sleutel) as Bereik;
        if (x.ks.indexOf(norm(slot.lokaal)) < 0) {
            x.ks.push(norm(slot.lokaal));
        }
        x.a = Math.min(x.a, slot.t);
        x.e = Math.max(x.e, slot.t + duur);
    }
    const gekozen: Map<string, string[]> = new Map<string, string[]>();
    bereik.forEach((x: Bereik, sleutel: string) => {
        const geschikt: string[] = [];
        for (const y of simus) {
            if ((y.d === null || y.d === x.d) && (y.van === null || y.van <= x.a)
                && (y.tot === null || y.tot >= x.e) && geschikt.indexOf(y.naam) < 0) {
                geschikt.push(y.naam);
            }
        }
        if (geschikt.length < x.ks.length) {
            meldingen.push(["Waarschuwing", `${datumTekst(x.d)} ${tijdTekst(x.a)}-${tijdTekst(x.e)}: ${geschikt.length} ` +
                `simucliënt(en) beschikbaar voor ${x.ks.length} lokalen; niet elk lokaal kan dan draaien.`]);
        }
        gekozen.set(sleutel, geschikt);
    });
    const uit: Map<string, Persoon[]> = new Map<string, Persoon[]>();
    for (const slot of slots) {
        const b: number = blok(slot.t);
        const g: string[] = gekozen.get(`${slot.d}_${b}`) as string[];
        if (g.length > 0) {
            const k: number = ((kamers.get(slot.d) as string[]).indexOf(norm(slot.lokaal)) + (rotatie ? b : 0)) % g.length;
            uit.set(`${slot.d}_${slot.t}_${norm(slot.lokaal)}`,
                g.slice(k).concat(g.slice(0, k)).map((naam: string) => ({ naam: naam, casus: "", lokaal: "" })));
        }
    }
    return uit;
}

// Per lokaal (toetsdag/simu-dagblad) of per tijd (oude bladen).
function kandidaten(beschikbaar: Map<string, Persoon[]>, slot: Slot): Persoon[] {
    const perLokaal: Persoon[] | undefined = beschikbaar.get(`${slot.d}_${slot.t}_${norm(slot.lokaal)}`);
    if (perLokaal && perLokaal.length > 0) {
        return perLokaal;
    }
    return beschikbaar.get(`${slot.d}_${slot.t}`) || [];
}

function maakRooster(bladen: Map<string, Matrix>, modus: string): Resultaat {
    const m: Modus = MODI.get(modus) as Modus;
    const fouten: string[] = [];
    const meldingen: Cel[][] = [];
    const inst: Inst = leesInstellingen(bladen.get(INSTELLINGEN_BLAD), fouten);
    const s: Map<string, Cel> = inst.w;
    const pz: number[][] = inst.pauzes;
    const txt = (k: string): string => s.get(k) as string;
    const num = (k: string): number => s.get(k) as number;
    const roosterBlad: string = txt(m.rooster);
    const meldBlad: string = txt("blad_meldingen");
    const uit: Map<string, Matrix> = new Map<string, Matrix>();
    const aangevuld: Matrix | null = instellingenAangevuld(bladen.get(INSTELLINGEN_BLAD));
    if (aangevuld) {
        uit.set(INSTELLINGEN_BLAD, aangevuld);
    }
    const studenten: Student[] = leesStudenten(bladen.get(txt("blad_studenten")), txt("blad_studenten"), fouten, meldingen);
    const dag: Matrix | undefined = bladen.get(txt(m.dag));
    const oudeSlots: Matrix | undefined = bladen.get(txt(m.slots));
    const dagmodus: boolean = heeftData(dag) || !oudeSlots || oudeSlots.length === 0;
    let slots: Slot[] = [];
    let beoordelaars: Map<string, Persoon[]> = new Map<string, Persoon[]>();
    if (dagmodus && (!dag || dag.length === 0)) {
        fouten.push(`Blad '${txt(m.dag)}' ontbreekt of is leeg.`);
    } else if (dagmodus) {
        const tijden: number[] = tijdgrid(num("start"), num("eind"), num("duur"), num("wissel"), pz);
        leesToetsdag(dag as Matrix, txt(m.dag), tijden, fouten, slots, beoordelaars);
    } else {
        slots = leesSlots(oudeSlots, txt(m.slots), fouten);
        beoordelaars = leesBeschikbaar(bladen.get(txt("blad_beo")), txt("blad_beo"), ["beoordelaar", "naam"], fouten);
    }
    const simuBlad: Matrix | undefined = bladen.get(txt("blad_simu"));
    let simuDag: SimuDag[] | null = null;
    let simus: Map<string, Persoon[]> = new Map<string, Persoon[]>();
    if (simuBlad && simuBlad.length > 0 && kolom(simuBlad[0], ["tijd"]) < 0) {
        simuDag = leesSimuDag(simuBlad, txt("blad_simu"), fouten);
    } else {
        simus = leesBeschikbaar(simuBlad, txt("blad_simu"), ["simuclint", "simu", "simulant", "naam"], fouten);
    }
    if (fouten.length > 0) {
        const rijen: Matrix = [MELDINGEN_KOP as Cel[]];
        fouten.forEach((f: string) => rijen.push(["Fout", f]));
        meldingen.forEach((x: Cel[]) => rijen.push(x));
        rijen.push(["Info", `${m.naam}: rooster NIET aangepast door bovenstaande fouten.`]);
        uit.set(meldBlad, rijen);
        return { fouten: fouten, bladen: uit, rooster: roosterBlad, meldingen: meldBlad };
    }

    const duurNormaal: number = num("duur");
    const duurTtv: number = num("ttv");
    const pauzenr = (t: number): number => pz.filter((p: number[]) => p[1] <= t).length;
    const regel: string = txt("casus_na");
    const wissel = (nr: number): boolean => regel !== "Nee" &&
        (regel === "Na elke pauze afwisselen" ? nr % 2 === 1 : nr >= Number(regel.charAt(regel.length - 1)));
    if (dagmodus) {
        meldingen.push(["Info", `Slots gemaakt uit blad '${txt(m.dag)}': ${slots.length} slots ` +
            `(${tijdTekst(num("start"))}-${tijdTekst(num("eind"))}, elke ${duurNormaal + num("wissel")} min).`]);
    }
    const bruikbaar: Slot[] = [];
    let inPauze: number = 0;
    const gelijk: string[] = [];
    for (const slot of slots) {
        if (pz.some((p: number[]) => p[0] <= slot.t && slot.t < p[1])) {
            inPauze++;
            continue;
        }
        if (slot.casusNa && norm(slot.casusNa) === norm(slot.casus) && gelijk.indexOf(`${slot.d}_${slot.lokaal}`) < 0) {
            gelijk.push(`${slot.d}_${slot.lokaal}`);
            meldingen.push(["Waarschuwing", `Lokaal ${slot.lokaal} op ${datumTekst(slot.d)}: ` +
                `'Casus na pauze' is gelijk aan 'Casus' (${slot.casus}).`]);
        }
        if (slot.casusNa && wissel(pauzenr(slot.t))) {
            slot.casus = slot.casusNa;
        }
        bruikbaar.push(slot);
    }
    if (inPauze > 0) {
        meldingen.push(["Info", `${inPauze} slot(s) vallen in een pauze en zijn overgeslagen.`]);
    }
    if (simuDag !== null) {
        simus = simuRotatie(bruikbaar, simuDag, pz, duurNormaal, txt("simu_rotatie") === "Ja", meldingen);
    }

    const doel: Student[] = studenten.filter((st: Student) => st.her === (modus === "her"));
    const groep = (st: Student): number => (txt("ttv_eerst") === "Ja" && st.ttv ? 0 : st.voorkeur ? 1 : 2);
    const volgorde: Student[] = doel.slice().sort((a: Student, b: Student) => vergelijk([groep(a), a.i], [groep(b), b.i]));
    const gewoon: Slot[] = bruikbaar.slice().sort((a: Slot, b: Slot) => vergelijk([a.d, a.t, a.n], [b.d, b.t, b.n]));
    const mid: number = num("middag");
    const sorteer = (sleutel: (x: Slot) => number[]): Slot[] =>
        bruikbaar.slice().sort((a: Slot, b: Slot) => vergelijk(sleutel(a), sleutel(b)));
    const perVoorkeur: Map<string, Slot[]> = new Map<string, Slot[]>([
        ["", gewoon], ["Vroeg", gewoon],
        ["Ochtend", sorteer((x: Slot) => [x.d, x.t < mid ? 0 : 1, x.t, x.n])],
        ["Middag", sorteer((x: Slot) => [x.d, x.t >= mid ? 0 : 1, x.t, x.n])],
        ["Laat", sorteer((x: Slot) => [x.d, -x.t, x.n])],
    ]);
    // Eind van blok vóór pauze: hoe dichter een slot bij de volgende pauze begint, hoe eerder.
    const totPauze = (t: number): number => Math.min(...pz.filter((p: number[]) => p[0] > t).map((p: number[]) => p[0] - t).concat([1000000]));
    const ttvVolgorde: Slot[] = txt("ttv_plek") === "Eind van blok vóór pauze"
        ? sorteer((x: Slot) => [x.d, totPauze(x.t), -x.t, x.n]) : gewoon;
    const deels: boolean = txt("deelnaam") === "Ja";
    const bezet: Map<string, number[][]> = new Map<string, number[][]>();

    const vrij = (sleutel: string, a: number, b: number): boolean =>
        (bezet.get(sleutel) || []).every((xy: number[]) => b <= xy[0] || a >= xy[1]);

    // Moet het anders dan bij de 1e kans? Verplicht altijd, Voorkeur alleen in de eerste (strikte) ronde.
    const anders = (r: string, strikt: boolean): boolean =>
        modus === "her" && (txt(r) === "Verplicht" || (strikt && txt(r) === "Voorkeur"));

    const kies = (lijst: Persoon[], slot: Slot, a: number, b: number, soort: string, st: Student, strikt: boolean): string | null => {
        for (const k of lijst) {
            if (k.casus && norm(k.casus) !== norm(slot.casus)) {
                continue;
            }
            if (k.lokaal && norm(k.lokaal) !== norm(slot.lokaal)) {
                continue;
            }
            if (soort === "b" && beperkt(k.naam, st.verboden, deels)) {
                continue;
            }
            if (soort === "b" && anders("her_beo", strikt) && st.eb && norm(k.naam) === norm(st.eb)) {
                continue;
            }
            if (vrij(`${soort}\u0001${slot.d}\u0001${k.naam}`, a, b)) {
                return k.naam;
            }
        }
        return null;
    };

    // Gewone toets niet in een pauze; TTV mag doorlopen in de pauze direct erna. Toetsdag: niet na de eindtijd.
    const past = (slot: Slot, a: number, b: number, ttv: boolean): boolean => {
        if (pz.some((p: number[]) => a < p[0] && b > (ttv ? p[1] : p[0]))) {
            return false;
        }
        return !(dagmodus && b > num("eind")) && vrij(`l\u0001${slot.d}\u0001${slot.lokaal}`, a, b);
    };

    const boek = (slot: Slot, a: number, b: number, st: Student, strikt: boolean): Toewijzing | null => {
        const beo: string | null = kies(kandidaten(beoordelaars, slot), slot, a, b, "b", st, strikt);
        const simu: string | null = beo ? kies(kandidaten(simus, slot), slot, a, b, "s", st, strikt) : null;
        if (!beo || !simu) {
            return null;
        }
        for (const k of [`l\u0001${slot.d}\u0001${slot.lokaal}`, `b\u0001${slot.d}\u0001${beo}`, `s\u0001${slot.d}\u0001${simu}`]) {
            if (!bezet.has(k)) {
                bezet.set(k, []);
            }
            (bezet.get(k) as number[][]).push([a, b]);
        }
        return { st: st, slot: slot, eind: b, beo: beo, simu: simu };
    };

    const probeer = (st: Student, strikt: boolean): Toewijzing | null => {
        const duur: number = st.ttv ? duurTtv : duurNormaal;
        for (const slot of (st.ttv ? ttvVolgorde : perVoorkeur.get(st.voorkeur) as Slot[])) {
            const a: number = slot.t;
            const b: number = slot.t + duur;
            if (!past(slot, a, b, st.ttv)) {
                continue;
            }
            if (anders("her_casus", strikt) && st.ec && norm(slot.casus) === norm(st.ec)) {
                continue;
            }
            const r: Toewijzing | null = boek(slot, a, b, st, strikt);
            if (r) {
                return r;
            }
        }
        return null;
    };

    const toegewezen: Map<number, Toewijzing> = new Map<number, Toewijzing>();
    for (const strikt of (modus === "her" ? [true, false] : [false])) {
        for (const st of volgorde) {
            if (!toegewezen.has(st.i)) {
                const r: Toewijzing | null = probeer(st, strikt);
                if (r) {
                    toegewezen.set(st.i, r);
                }
            }
        }
    }

    const vrije: Toewijzing[] = [];
    if (txt("vrije_plekken") === "Ja") {
        const niemand: Student = {
            i: -1, nr: "", voornaam: "", tv: "", achternaam: "", klas: "", verboden: [], ttv: false, her: false,
            eb: "", ec: "", voorkeur: "", opm: "",
        };
        for (const slot of gewoon) {
            const a: number = slot.t;
            const b: number = slot.t + duurNormaal;
            const r: Toewijzing | null = past(slot, a, b, false) ? boek(slot, a, b, niemand, false) : null;
            if (r) {
                r.st = null;
                vrije.push(r);
            }
        }
    }

    const roosterSleutel = (r: Toewijzing): (string | number)[] => txt("rooster_sort") === "Lokaal, dan tijd"
        ? [r.slot.d, r.slot.lokaal.toLowerCase(), r.slot.t, r.slot.n]
        : [r.slot.d, r.slot.t, r.slot.lokaal.toLowerCase(), r.slot.n];
    const uitdraaiSleutel = (r: Toewijzing): (string | number)[] => {
        const st: Student = r.st as Student;
        if (txt("uitdraai_sort") === "Achternaam") {
            return [st.achternaam.toLowerCase(), st.voornaam.toLowerCase(), st.i];
        }
        if (txt("uitdraai_sort") === "Tijd") {
            return [r.slot.d, r.slot.t, st.achternaam.toLowerCase(), st.i];
        }
        return [r.slot.d, r.slot.lokaal.toLowerCase(), r.slot.t, st.i];
    };

    const lijst: Toewijzing[] = [];
    toegewezen.forEach((r: Toewijzing) => lijst.push(r));
    const rooster: Matrix = [ROOSTER_KOP as Cel[]];
    const alles: Toewijzing[] = lijst.concat(vrije);
    alles.sort((x: Toewijzing, y: Toewijzing) => vergelijk(roosterSleutel(x), roosterSleutel(y)));
    for (const r of alles) {
        const slot: Slot = r.slot;
        if (r.st === null) {
            rooster.push([slot.d, slot.t / 1440, slot.lokaal, slot.casus, slot.vaardigheid, r.beo, r.simu,
                "", "", "", "", "", VRIJ, r.eind / 1440]);
            continue;
        }
        const st: Student = r.st;
        const naam: string = `${st.voornaam} ${st.achternaam}`;
        const bijz: string[] = st.ttv ? [`TTV (${duurTtv} min)`] : [];
        if (modus === "her" && st.eb && norm(st.eb) === norm(r.beo)) {
            bijz.push("Zelfde beoordelaar als 1e kans");
            meldingen.push(["Waarschuwing", `${naam}: zelfde beoordelaar als 1e kans.`]);
        }
        if (modus === "her" && st.ec && norm(st.ec) === norm(slot.casus)) {
            bijz.push("Zelfde casus als 1e kans");
            meldingen.push(["Waarschuwing", `${naam}: zelfde casus als 1e kans.`]);
        }
        if ((st.voorkeur === "Ochtend" && slot.t >= mid) || (st.voorkeur === "Middag" && slot.t < mid)) {
            meldingen.push(["Waarschuwing", `${naam}: voorkeur ${st.voorkeur.toLowerCase()} niet gehaald (${tijdTekst(slot.t)}).`]);
        }
        if (st.opm) {
            bijz.push(st.opm);
        }
        rooster.push([slot.d, slot.t / 1440, slot.lokaal, slot.casus, slot.vaardigheid, r.beo, r.simu,
            st.nr, st.voornaam, st.tv, st.achternaam, st.klas, bijz.join("; "), r.eind / 1440]);
    }
    const uitdraai: Matrix = [UITDRAAI_KOP as Cel[]];
    lijst.sort((x: Toewijzing, y: Toewijzing) => vergelijk(uitdraaiSleutel(x), uitdraaiSleutel(y)));
    for (const r of lijst) {
        const st: Student = r.st as Student;
        const naam: string = [st.voornaam, st.tv, st.achternaam].filter((x: string) => x !== "").join(" ");
        const tijd: string = tijdTekst(r.slot.t);
        uitdraai.push([naam, r.slot.lokaal, txt("uitdraai_datum") === "Ja" ? `${datumTekst(r.slot.d)} ${tijd}` : tijd]);
    }
    const niet: Student[] = volgorde.filter((st: Student) => !toegewezen.has(st.i)).sort((a: Student, b: Student) => a.i - b.i);
    for (const st of niet) {
        rooster.push(["", "", "", "", "", "", "", st.nr, st.voornaam, st.tv, st.achternaam, st.klas, NIET, ""]);
        meldingen.push(["Waarschuwing", `${st.voornaam} ${st.achternaam} is NIET ingeroosterd: geen vrij lokaal, ` +
            "beoordelaar of simucliënt gevonden (let ook op uitsluitingen en verplichte herkansingsregels)."]);
    }
    if (vrije.length > 0) {
        meldingen.push(["Info", `${vrije.length} vrije plek(ken) over. Ze staan als '${VRIJ}' in het rooster, ` +
            "niet op de uitdraai of in de PDF's."]);
    }

    uit.set(roosterBlad, rooster);
    uit.set(txt(m.uitdraai), uitdraai);
    if (modus === "kans1") {
        const oud: Matrix = bladen.get(txt("blad_studenten")) as Matrix;
        const kop: Cel[] = oud[0].slice();
        const cols: number[] = EERSTE_KOP.map((naam: string) => {
            let c: number = kolom(kop, [norm(naam)]);
            if (c < 0) {
                c = kop.length;
                kop.push(naam);
            }
            return c;
        });
        const nieuw: Matrix = oud.map((rij: Cel[]) => rij.concat(new Array<Cel>(Math.max(0, kop.length - rij.length)).fill("")));
        nieuw[0] = kop;
        for (const st of doel) {
            const r: Toewijzing | undefined = toegewezen.get(st.i);
            const waarden: Cel[] = r ? [r.slot.d, r.slot.t / 1440, r.beo, r.slot.casus] : ["", "", "", ""];
            cols.forEach((c: number, j: number) => { nieuw[st.i][c] = waarden[j]; });
        }
        uit.set(txt("blad_studenten"), nieuw);
    }
    let info: string = `${m.naam}: ${toegewezen.size} van ${doel.length} studenten ingeroosterd (jaar ${s.get("jaar")}, toets ${duurNormaal} min, TTV ${duurTtv} min`;
    info += pz.length > 0 ? `, pauzes ${pauzesTekst(pz)}).` : ", geen pauze).";
    uit.set(meldBlad, [MELDINGEN_KOP as Cel[], ["Info", info] as Cel[]].concat(meldingen));
    return { fouten: [], bladen: uit, rooster: roosterBlad, meldingen: meldBlad };
}
